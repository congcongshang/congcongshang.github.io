connect.open_receiver({

// WebSocket 广播函数
function broadcastToClients(data) {
  wss.clients.forEach(client => {
    if (client.readyState === WebSocket.OPEN) {
      client.send(JSON.stringify(data));
    }
  });
}
container.on('connection_open', () => console.log('连接成功'));
container.on('error', (err) => console.error('AMQP 错误:', err));
// AMQP 消息处理
container.on('message', (context) => {
  try {
    const msg = context.message;
    const body = JSON.parse(`${msg.body}`);
    console.log(body.notify_data.body.services);
    broadcastToClients({
      deviceId: body.notify_data.header.device_id,
      timestamp: new Date(body.event_time_ms).getTime(),
      data: body.notify_data.body.services
    });
    context.delivery.accept();
  } catch (e) {
    context.delivery.release();
  }
});
// 断线重连机制
container.on('connection_close', () => {
  setTimeout(() => connect.open_receiver('DefaultQueue'), 5000);
});

5. 北向APP连接华为云

北向开发APP通过http请求华为云，直接使用HTTP请求的时候， 会需要一个认证鉴权， 因此在开始之前咱们需要先创建一个认证用户，并赋予该用户一定权限； 然后按照华为云认证鉴权的格式设置相应请求字符串，获取认证token； 最后再做HTTP请求时带着这个token就可以顺利请求到华为云了。

5.1. 创建认证用户

在统一身份认证中创建一个用户

让它拥有所有的权限

5.2. 获取认证token

首先定义出获取token字符串格式（这个不用变直接用）


class User {
  auth: Auth

  constructor(name: string, password: string, domain: string) {
    this.auth = {
      identity: {
        methods: ["password"],
        password: {
          user: {
            name: name,
            password: password,
            domain: {
              name: domain
            }
          }
        },
      },
      scope: {
        domain: {
          name: domain
        }
      }
    }
  }
}

interface Auth {
  identity: Identity
  scope: Scope
}

interface Identity {
  methods: string[]
  password: Password
}

interface IAMUser {
  name: string
  password: string
  domain: Domain
}

interface Password {
  user: IAMUser
}

interface Domain {
  name: string
}

interface Scope {
  domain: Domain
}

然后创建对象请求华为云touken（这个里面需要改：用户名，密码，账户名）

注意：用户名和密码就是统一身份认证创建的


import { http } from '@kit.NetworkKit'


  @State hw_token: string = ''

  async onPageShow() {
    await this.getToken()
    this.hw_token = AppStorage.get<string>('hw_token') ?? '获取失败'
  }

  /**
   * 认证鉴权
   */
  async getToken() {
    let url: string = 'https://iam.myhuaweicloud.com/v3/auth/tokens'
    let hw_user: User = new User('abc', '12345678', 'xyz_domain') // 改成自己的

    let httpRequest = http.createHttp()
    let options: http.HttpRequestOptions = {
      method: http.RequestMethod.POST,
      extraData: hw_user,
      expectDataType: http.HttpDataType.OBJECT
    }
    let response: http.HttpResponse = await httpRequest.request(url, options)
    console.log('token response:', JSON.stringify(response.header['x-subject-token']))

    // 放入应用级缓存
    AppStorage.setOrCreate<string>('hw_token', response.header['x-subject-token'])
  }
