1. 读写文件常用代码


def read_file(file_name, encoding="utf-8", skip_empty_line=True):
    """
    读取tsv文本，按\t分割为二维列表
    :param skip_empty_line: 是否跳过空行
    :return: list[list[str]]
    """
    result = []
    with open(file_name, "r", encoding=encoding) as f:
        for line in f:
            # 仅移除末尾换行，保留本行内所有\t、首尾空字段
            line = line.rstrip("\n")
            if skip_empty_line and line == "":
                continue
            row = line.split("\t")
            result.append(row)
    return result


def write_file(data, file_name, encoding="utf-8", skip_empty_row=True):
    """
    将二维列表写入tsv文本，字段用\t分隔，一行对应一个子列表
    :param data: 二维列表 list[list[str]]
    :param skip_empty_row: 是否跳过空行（子列表为空）
    :return: None
    """
    with open(file_name, "w", encoding=encoding, newline="") as f:
        for row in data:
            if skip_empty_row and len(row) == 0:
                continue
            line = "\t".join(row) + "\n"
            f.write(line)
