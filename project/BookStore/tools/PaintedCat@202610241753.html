1. ESM导入路径

学习node使用服务端技术，优点比较多，加油学习，顺利拿到想要的技术，还是有很多可以思考的事情呢！


// get-path.js
import { fileURLToPath } from 'url'
import { dirname, join } from 'path'
export default function (metaUrl) {
  const f = fileURLToPath(metaUrl)
  const d = dirname(f)
  return {
    __filename: f,
    __dirname: d,
    join: (...args) => join(d, ...args)
  }
}

使用：


import getPath from './get-path.js'
const { __dirname,__filename, join } = getPath(import.meta.url)
